cmake_minimum_required(VERSION 3.16)
project(xlerobot_control)
find_package(ament_cmake REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
add_library(xlerobot_sim_system SHARED src/sim_system.cpp)
target_compile_features(xlerobot_sim_system PUBLIC cxx_std_17)
ament_target_dependencies(xlerobot_sim_system hardware_interface pluginlib rclcpp rclcpp_lifecycle)
pluginlib_export_plugin_description_file(hardware_interface xlerobot_control.xml)
install(TARGETS xlerobot_sim_system LIBRARY DESTINATION lib)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
